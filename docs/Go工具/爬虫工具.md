# Go 爬虫工具（Sideloader）

本文档介绍 Sideloader 爬虫工具的实现和使用。

## 功能特性

- **网站爬取** - 递归抓取 BetterRepack 下载站
- **ZIP 流式解析** - 不下载完整文件即可提取 GUID
- **并发控制** - 限制并发数，避免被封禁
- **代理支持** - 支持 HTTP/HTTPS 代理
- **断点续抓** - 支持中断后从上次位置继续

## 使用方法

### 命令行参数

```bash
sideloader [选项]

选项:
  -proxy string    代理地址 (可选)
```

### 示例

```bash
# 基本使用
./sideloader

# 使用代理
./sideloader -proxy "http://127.0.0.1:7890"

# 带认证的代理
./sideloader -proxy "http://user:pass@127.0.0.1:7890"
```

## 代码实现

### 核心结构

```go
var (
  httpClient *http.Client
  baseURL    string
  result     = make(map[string]string)  // GUID -> URL
  resultMu   sync.Mutex
  modSem     chan struct{}              // Mod 处理信号量
```

### HTTP 客户端初始化

```go
func initHTTPClient(proxy string) {
  transport := &http.Transport{
    MaxIdleConns:        50,
    MaxIdleConnsPerHost: 10,
    IdleConnTimeout:     90 * time.Second,
  }

  if proxy != "" {
    p, _ := url.Parse(proxy)
    transport.Proxy = http.ProxyURL(p)
    fmt.Printf("[Sideloader] Using proxy: %s\n", proxy)
  }

  httpClient = &http.Client{
    Transport: transport,
    Timeout:   60 * time.Second,
  }
}
```

### 页面抓取

```go
func fetchDoc(pageURL string) (*goquery.Document, error) {
  req, err := http.NewRequest("GET", pageURL, nil)
  if err != nil {
    return nil, err
  }

  req.Header.Set("User-Agent", userAgent)

  resp, err := httpClient.Do(req)
  if err != nil {
    return nil, err
  }
  defer resp.Body.Close()

  if resp.StatusCode != http.StatusOK {
    return nil, fmt.Errorf("HTTP %d", resp.StatusCode)
  }

  return goquery.NewDocumentFromReader(resp.Body)
}
```

### 链接解析

```go
func parseLinks(doc *goquery.Document, pageURL string) (dirs, mods []string) {
  hrefMap := make(map[string]bool)

  doc.Find("table#indexlist tr").Each(func(i int, s *goquery.Selection) {
    if i <= 1 {  // 跳过表头
      return
    }

    s.Find("a").Each(func(_ int, a *goquery.Selection) {
      if href, ok := a.Attr("href"); ok {
        if decoded, err := url.QueryUnescape(href); err == nil &&
           decoded != "" && decoded != "/" {
          hrefMap[pageURL+decoded] = true
        }
      }
    })
  })

  for href := range hrefMap {
    if strings.HasSuffix(strings.ToLower(href), ".zipmod") {
      mods = append(mods, href)
    } else if strings.HasSuffix(href, "/") {
      dirs = append(dirs, href)
    }
  }

  return dirs, mods
}
```

### ZIP 流式解析

核心功能：不下载整个 zipmod 文件，仅读取尾部 Central Directory 提取 manifest.xml：

```go
func extractGUIDFromZipmod(modURL string) string {
  size, err := getFileSize(modURL)
  if err != nil {
    return ""
  }

  var data []byte

  if size <= 65536 {
    // 小文件：直接下载
    data, _ = fetchRange(modURL, 0, size-1)
  } else {
    // 大文件：渐进式读取尾部（每次 16KB，最多 256KB）
    const chunkSize = int64(16384)
    for offset := int64(0); offset < 262144 && offset < size; offset += chunkSize {
      end := size - offset - 1
      start := end - chunkSize + 1
      if start < 0 {
        start = 0
      }

      chunk, _ := fetchRange(modURL, start, end)
      data = append(chunk, data...)

      // 尝试解析 Central Directory
      if entries := readCentralDir(data, size); entries != nil {
        return extractFromEntries(entries, modURL)
      }
    }
  }

  if entries := readCentralDir(data, size); entries != nil {
    return extractFromEntries(entries, modURL)
  }

  return ""
}

func fetchRange(modURL string, start, end int64) ([]byte, error) {
  req, _ := http.NewRequest("GET", modURL, nil)
  req.Header.Set("User-Agent", userAgent)
  req.Header.Set("Range", fmt.Sprintf("bytes=%d-%d", start, end))
  req.Header.Set("Accept-Encoding", "identity")

  resp, err := httpClient.Do(req)
  if err != nil {
    return nil, err
  }
  defer resp.Body.Close()

  return io.ReadAll(resp.Body)
}

func getFileSize(modURL string) (int64, error) {
  req, _ := http.NewRequest("HEAD", modURL, nil)
  req.Header.Set("User-Agent", userAgent)

  resp, err := httpClient.Do(req)
  if err != nil {
    return 0, err
  }
  defer resp.Body.Close()

  return strconv.ParseInt(resp.Header.Get("Content-Length"), 10, 64)
}
```

### Manifest 提取

```go
func extractFromEntries(entries map[string]fileEntry, modURL string) string {
  var entry *fileEntry

  // 查找 manifest.xml
  for name, e := range entries {
    if strings.EqualFold(name, "manifest.xml") {
      entry = &e
      break
    }
  }

  if entry == nil {
    return ""
  }

  // 获取 manifest.xml 的数据偏移
  headerData, _ := fetchRange(modURL,
    int64(entry.offset),
    int64(entry.offset)+200)

  dataOffset, compressedSize, err := parseLocalHeader(headerData)
  if err != nil {
    dataOffset = int64(30 + len(entry.name))
    compressedSize = entry.compressedSize
  }

  // 读取 manifest.xml 内容
  actualOffset := int64(entry.offset) + dataOffset
  manifestData, _ := fetchRange(modURL,
    actualOffset,
    actualOffset+int64(compressedSize)-1)

  // 解压并解析 XML
  guid, _ := extractManifestXML(manifestData, entry.compressionMethod)
  return guid
}
```

### 并发爬取

```go
func crawl(pageURL string) {
  fmt.Printf("[Sideloader] Processing: %s\n", pageURL)

  doc, err := fetchDoc(pageURL)
  if err != nil {
    return
  }

  dirs, mods := parseLinks(doc, pageURL)

  // 并发处理 Mod 文件
  var wg sync.WaitGroup
  for _, mod := range mods {
    wg.Add(1)
    modSem <- struct{}{}

    go func(modURL string) {
      defer wg.Done()
      defer func() { <-modSem }()

      if guid := extractGUIDFromZipmod(modURL); guid != "" {
        resultMu.Lock()
        result[guid] = strings.TrimPrefix(modURL, baseURL)
        resultMu.Unlock()
        fmt.Printf("[Sideloader] Found: %s\n", guid)
      }
    }(mod)
  }
  wg.Wait()

  // 并发处理子目录
  dirSem := make(chan struct{}, 3)
  var dirWg sync.WaitGroup

  for _, dir := range dirs {
    dirWg.Add(1)
    dirSem <- struct{}{}

    go func(dirURL string) {
      defer dirWg.Done()
      defer func() { <-dirSem }()
      crawl(dirURL)
    }(dir)
  }
  dirWg.Wait()
}
```

### 主函数

```go
func main() {
  var proxy string
  for i := 1; i < len(os.Args); i++ {
    if os.Args[i] == "-proxy" && i+1 < len(os.Args) {
      proxy = os.Args[i+1]
      i++
    }
  }

  initHTTPClient(proxy)

  // 初始化信号量
  modSem = make(chan struct{}, 10)

  // 开始爬取
  baseURL = "https://sideload.betterrepack.com/download/AISHS2/"
  fmt.Printf("[Sideloader] Starting crawl from %s\n", baseURL)
  crawl(baseURL)

  fmt.Printf("[Sideloader] Crawl completed, found %d mods\n", len(result))

  // 保存结果
  file, err := os.Create("result.json")
  if err != nil {
    fmt.Printf("[Sideloader] Failed to create result.json: %v\n", err)
    os.Exit(1)
  }
  defer file.Close()

  encoder := json.NewEncoder(file)
  encoder.SetIndent("", "  ")
  if err := encoder.Encode(result); err != nil {
    fmt.Printf("[Sideloader] Failed to encode result: %v\n", err)
    os.Exit(1)
  }

  fmt.Printf("[Sideloader] Saved to result.json\n")
}
```

## 与主进程集成

```typescript
// src/main/goSideloader.ts
import { spawn, ChildProcess } from 'child_process'

let sideloaderProcess: ChildProcess | null = null

export const runSideloader = (proxy?: string): Promise<void> => {
  return new Promise((resolve, reject) => {
    const args: string[] = []
    if (proxy) {
      args.push('-proxy', proxy)
    }

    sideloaderProcess = spawn(sideloaderPath, args)

    sideloaderProcess.stdout?.on('data', (data) => {
      console.log(`[Sideloader] ${data}`)
    })

    sideloaderProcess.stderr?.on('data', (data) => {
      console.log(`[Sideloader] ${data}`)
    })

    sideloaderProcess.on('close', (code) => {
      sideloaderProcess = null
      if (code === 0) {
        resolve()
      } else {
        reject(new Error(`Sideloader exited with code ${code}`))
      }
    })

    sideloaderProcess.on('error', (err) => {
      reject(err)
    })
  })
}

export const stopSideloader = (): void => {
  if (sideloaderProcess) {
    sideloaderProcess.kill('SIGTERM')
    sideloaderProcess = null
  }
}

export const isSideloaderRunning = (): boolean => {
  return sideloaderProcess !== null && !sideloaderProcess.killed
}

export const checkSideloaderExists = (): boolean => {
  return fs.existsSync(sideloaderPath)
}
```

## ZIP 格式解析原理

```
ZIP 文件结构:
┌─────────────────────────────────────┐
│ [Local File Header 1]               │
│ [File Data 1]                       │
│ [Data Descriptor 1]                 │
├─────────────────────────────────────┤
│ [Local File Header 2]               │
│ [File Data 2]                       │
│ [Data Descriptor 2]                 │
├─────────────────────────────────────┤
│ ... 更多文件 ...                     │
├─────────────────────────────────────┤
│ [Central Directory Header 1]        │  ← 从文件尾部开始读取
│ [Central Directory Header 2]        │
│ ...                                 │
│ [End of Central Directory Record]   │  ← 固定格式，可快速定位
└─────────────────────────────────────┘

优化策略:
1. 通过 HTTP Range 请求只读取尾部
2. 找到 End of Central Directory Record
3. 定位 Central Directory 位置
4. 解析所有文件条目
5. 找到 manifest.xml 的偏移
6. 再次 Range 请求读取 manifest.xml
```

## 注意事项

1. **网站限制** - 遵守网站的 robots.txt 和访问频率限制
2. **网络稳定** - 不稳定的网络可能导致解析失败
3. **文件损坏** - 损坏的 zipmod 文件无法解析
4. **内存控制** - 大文件通过流式读取避免内存溢出
