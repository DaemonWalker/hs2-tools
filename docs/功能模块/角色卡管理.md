# 角色卡管理

本文档介绍角色卡浏览和管理功能。

## 功能概述

角色卡管理功能允许用户：

- 浏览本地女性/男性角色卡（无限滚动加载）
- 查看角色卡缩略图（支持加载失败占位图）
- 按多种方式排序（名称、路径、收藏优先）
- 收藏角色卡（本地存储）
- 查看角色卡所需的 Mod 列表
- 在文件管理器中定位角色卡文件
- 管理角色卡（移动、删除等）

## 页面路由

| 路由 | 页面 | 说明 |
|------|------|------|
| `/character` | 女性角色卡 | 显示女性角色卡列表 |
| `/character/male` | 男性角色卡 | 显示男性角色卡列表 |

### 角色卡数据结构


### 排序类型

```typescript
type SortType = 'nameAsc' | 'nameDesc' | 'path' | 'favorite'

// 排序方式说明：
// - nameAsc: 名称 A-Z（按文件名排序）
// - nameDesc: 名称 Z-A（按文件名倒序）
// - path: 路径排序（按完整路径排序）
// - favorite: 收藏优先（默认）
//   * 收藏的角色按收藏顺序排在前面
//   * 未收藏的角色按名称 A-Z 排序
```

### 2. 无限滚动卡片网格 (CardGrid)

使用 ahooks useInfiniteScroll 实现无限滚动：

```typescript
// wails/frontend/src/components/share/CardGrid.tsx
import { useInfiniteScroll } from 'ahooks'

export const CardGrid = forwardRef<CardGridRef, IProps>(({
  type,
  onCardClick,
  selectedPath,
  onTotalChange,
  onLoadingChange,
  sortType = 'favorite'
}, ref) => {
  const [favorites, setFavorites] = useState<string[]>(getFavorites)
  const sortedPathsRef = useRef<string[]>([])

  // 排序逻辑
  const sortPaths = useCallback((paths: string[], type: SortType): string[] => {
    const sorted = [...paths]
    
    switch (type) {
      case 'nameAsc':
        return sorted.sort((a, b) => {
          const nameA = a.split(/[\\/]/).pop()?.replace(/\.png$/i, '') || ''
          const nameB = b.split(/[\\/]/).pop()?.replace(/\.png$/i, '') || ''
          return nameA.localeCompare(nameB, 'zh-CN')
        })
      
      case 'nameDesc':
        return sorted.sort((a, b) => {
          const nameA = a.split(/[\\/]/).pop()?.replace(/\.png$/i, '') || ''
          const nameB = b.split(/[\\/]/).pop()?.replace(/\.png$/i, '') || ''
          return nameB.localeCompare(nameA, 'zh-CN')
        })
      
      case 'path':
        return sorted.sort((a, b) => a.localeCompare(b, 'zh-CN'))
      
      case 'favorite':
      default:
        // 收藏优先排序
        return sorted.sort((a, b) => {
          const indexA = favorites.indexOf(a)
          const indexB = favorites.indexOf(b)
          
          const isFavA = indexA !== -1
          const isFavB = indexB !== -1
          
          if (isFavA && isFavB) {
            return indexA - indexB
          } else if (isFavA) {
            return -1
          } else if (isFavB) {
            return 1
          } else {
            const nameA = a.split(/[\\/]/).pop()?.replace(/\.png$/i, '') || ''
            const nameB = b.split(/[\\/]/).pop()?.replace(/\.png$/i, '') || ''
            return nameA.localeCompare(nameB, 'zh-CN')
          }
        })
    }
  }, [favorites])

  // 无限滚动
  const { data, loading, loadingMore, reload, noMore, mutate } = useInfiniteScroll(
    async (currentData) => {
      let paths = sortedPathsRef.current
      if (paths.length === 0) {
        const rawPaths = await getAllFiles(charaPath!, { targetExtension: ['.png'] })
        paths = sortPaths(rawPaths, sortType)
        sortedPathsRef.current = paths
      }
      
      // 分批加载
      const currentCount = currentData?.list.length || 0
      const nextBatch = paths.slice(currentCount, currentCount + BATCH_SIZE)
      const results = await readPngPageDataBatch(nextBatch)
      
      const newItems = results.map(item => ({
        path: item.path,
        names: item.names,
        imageData: item.imageData
      }))
      
      return { 
        list: newItems, 
        hasMore: currentCount + newItems.length < paths.length, 
        total: paths.length 
      }
    },
    { target: containerRef, isNoMore: (data) => !data?.hasMore }
  )

  // 排序类型变化时重新加载
  useEffect(() => {
    if (charaPath && sortedPathsRef.current.length > 0) {
      sortedPathsRef.current = []
      mutate({ list: [], hasMore: true, total: 0 })
    }
  }, [sortType, charaPath])

  return (
    <div ref={containerRef} className="card-grid-container">
      <div className="card-grid" style={{
        display: 'grid',
        gridTemplateColumns: 'repeat(auto-fill, minmax(180px, 1fr))',
        gap: 16,
      }}>
        {items.map((item, index) => (
          <CardItem
            key={item.path}
            data={item}
            index={index}
            selected={selectedPath === item.path}
            isFavorite={favorites.includes(item.path)}
            onClick={onCardClick}
            onToggleFavorite={toggleFavorite}
          />
        ))}
      </div>
      
      {loadingMore && <div>加载中...</div>}
      {noMore && <Divider>已加载全部 {total} 个角色</Divider>}
    </div>
  )
})
```

### 3. 收藏功能

收藏状态使用 localStorage 持久化存储：

```typescript
// 收藏数据存储 key
const FAVORITE_KEY = 'chara_favorites'

// 从 localStorage 读取收藏列表
const getFavorites = (): string[] => {
  try {
    const data = localStorage.getItem(FAVORITE_KEY)
    return data ? JSON.parse(data) : []
  } catch {
    return []
  }
}

// 保存收藏列表到 localStorage
const saveFavorites = (favorites: string[]) => {
  localStorage.setItem(FAVORITE_KEY, JSON.stringify(favorites))
}

// 切换收藏状态
const toggleFavorite = (path: string) => {
  setFavorites(prev => {
    const index = prev.indexOf(path)
    let newFavorites: string[]
    
    if (index === -1) {
      // 添加收藏（放到最前面）
      newFavorites = [path, ...prev]
    } else {
      // 取消收藏
      newFavorites = prev.filter(p => p !== path)
    }
    
    saveFavorites(newFavorites)
    return newFavorites
  })
}
```

```go
func (a *App) OpenInFolder(filePath string) error {
    var cmd *exec.Cmd
    switch runtime.GOOS {
    case "windows":
        cmd = exec.Command("explorer", "/select,", filePath)
    case "darwin":
        cmd = exec.Command("open", "-R", filePath)
    default:
        cmd = exec.Command("xdg-open", filepath.Dir(filePath))
    }
    return cmd.Start()
}
```

```typescript
// wails/frontend/src/store/charaStore.ts
interface CharaState {
  cards: CharaCardInfo[]
  selectedCard: CharaCardInfo | null
  loading: boolean
  sortType: SortType
  favorites: string[]

  setCards: (cards: CharaCardInfo[]) => void
  selectCard: (card: CharaCardInfo) => void
  setSortType: (type: SortType) => void
  toggleFavorite: (path: string) => void
}

export const useCharaStore = create<CharaState>((set) => ({
  cards: [],
  selectedCard: null,
  loading: false,
  sortType: 'favorite',
  favorites: getFavorites(),

  setCards: (cards) => set({ cards }),
  selectCard: (card) => set({ selectedCard: card }),
  setSortType: (type) => set({ sortType: type }),
  toggleFavorite: (path) => set((state) => {
    const index = state.favorites.indexOf(path)
    const newFavorites = index === -1
      ? [path, ...state.favorites]
      : state.favorites.filter(p => p !== path)
    saveFavorites(newFavorites)
    return { favorites: newFavorites }
  })
}))
```

## 注意事项

1. **无限滚动** - 使用 `useInfiniteScroll` 实现，每批加载 24 个
2. **排序缓存** - 排序后的路径列表使用 `useRef` 缓存，避免重复计算
3. **收藏存储** - 收藏状态使用 localStorage 持久化，key 为 `chara_favorites`
4. **图片错误** - 缩略图加载失败时显示"暂无预览"占位图
5. **Mod 检测** - 扫描本地 Mods 后缓存结果，避免重复计算
6. **抽屉高度** - 从 50% 调整为 70%，支持点击外部关闭
7. **卡片宽度** - 固定 180px，使用 CSS Grid 自动计算列数
