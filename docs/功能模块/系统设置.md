# 系统设置

本文档介绍系统设置功能。

- 游戏路径
- 代理设置
- 防休眠选项
- 其他应用设置

| 路由 | 页面 | 说明 |
|------|------|------|
| `/settings` | 系统设置 | 应用配置页面 |

## 核心功能实现

### 1. 设置表单

```typescript
// src/renderer/src/pages/SystemSettings.tsx
interface SettingsForm {
  path: string
  proxyUri: string
  proxyUsername: string
  proxyPassword: string
  disableSleep: boolean
}

export const SystemSettings: FC = () => {
  const [form] = Form.useForm<SettingsForm>()
  const { settings, setSettingForm } = useSettingStore()

  useEffect(() => {
    form.setFieldsValue({
      path: settings.path || '',
      proxyUri: settings.proxy?.uri || '',
      proxyUsername: settings.proxy?.username || '',
      proxyPassword: settings.proxy?.password || '',
      disableSleep: settings.windowsSleep?.disabled || false
    })
  }, [settings])

  const handleSave = async (values: SettingsForm) => {
    await setSettingForm({
      path: values.path,
      proxy: {
        uri: values.proxyUri,
        username: values.proxyUsername,
        password: values.proxyPassword
      },
      windowsSleep: {
        disabled: values.disableSleep
      }
    })

    // 应用代理设置
    if (values.proxyUri) {
      await window.api.setProxy({
        uri: values.proxyUri,
        username: values.proxyUsername,
        password: values.proxyPassword
      })
    }

    message.success('设置已保存')
  }

  return (
    <div className="settings-page">
      <Form
        form={form}
        layout="vertical"
        onFinish={handleSave}
      >
        <Card title="游戏路径设置">
          <Form.Item name="path" label="游戏路径">
            <Input readOnly addonAfter={
              <Button onClick={handleSelectPath}>选择路径</Button>
            } />
          </Form.Item>
        </Card>

        <Card title="代理设置">
          <Form.Item name="proxyUri" label="代理地址">
            <Input placeholder="http://127.0.0.1:7890" />
          </Form.Item>
          <Form.Item name="proxyUsername" label="用户名">
            <Input />
          </Form.Item>
          <Form.Item name="proxyPassword" label="密码">
            <Input.Password />
          </Form.Item>
        </Card>

        <Card title="系统选项">
          <Form.Item name="disableSleep" valuePropName="checked">
            <Checkbox>下载时防止系统休眠</Checkbox>
          </Form.Item>
        </Card>

        <Button type="primary" htmlType="submit">
          保存设置
        </Button>
      </Form>
    </div>
  )
}
```

```typescript
// src/main/ipcHandler.ts
setProxy: (_, proxy: ProxyInfo) => {
  const proxyOptions: ProxyAgent.Options = { uri: proxy.uri || '' }

  if (proxy.username && proxy.password) {
    proxyOptions.token = `Basic ${Buffer.from(
      `${proxy.username}:${proxy.password}`
    ).toString('base64')}`
  }

  setGlobalDispatcher(new ProxyAgent(proxyOptions))

  // 保存供 Go 工具使用
  currentProxyUri = proxy.uri
  currentProxyAuth = proxy.username
    ? `${proxy.username}:${proxy.password}`
    : undefined
}
```

## 状态管理

```typescript
// src/renderer/src/store/settingStore.ts
interface SettingState {
  settings: SettingModel
  setPath: (path: string) => Promise<void>
  setSettingForm: (form: Partial<SettingWindowForm>) => Promise<void>
  init: () => Promise<void>

  // 派生路径
  scenePath: () => string
  charaFemalePath: () => string
  modsPath: () => string
}

export const useSettingStore = create<SettingState>((set, get) => ({
  settings: { path: undefined },

  init: async () => {
    const settings = await window.api.loadSettings()
    set({ settings: settings || { path: undefined } })
  },

  setPath: async (path) => {
    const settings = { ...get().settings, path }
    await window.api.saveSettings(settings)
    set({ settings })
  },

  setSettingForm: async (form) => {
    const current = get().settings
    const settings = {
      ...current,
      ...form,
      proxy: form.proxy ? { ...current.proxy, ...form.proxy } : current.proxy
    }
    await window.api.saveSettings(settings)
    set({ settings })
  },

  scenePath: () => get().settings.path + SCENE_PATH,
  charaFemalePath: () => get().settings.path + CHARA_FEMALE_PATH,
  modsPath: () => get().settings.path + MODS_PATH
}))
```

## 数据导入导出

```typescript
// 导出设置和数据
const exportData = async () => {
  const data = {
    settings: await window.api.loadSettings(),
    mods: await window.api.loadLocalMods(),
    exportedAt: new Date().toISOString()
  }

  const blob = new Blob([JSON.stringify(data, null, 2)], {
    type: 'application/json'
  })

  const url = URL.createObjectURL(blob)
  const a = document.createElement('a')
  a.href = url
  a.download = `hs2-tools-backup-${formatDate(new Date())}.json`
  a.click()
  URL.revokeObjectURL(url)
}

// 导入设置和数据
const importData = async (file: File) => {
  const text = await file.text()
  const data = JSON.parse(text)

  if (data.settings) {
    await window.api.saveSettings(data.settings)
  }
  if (data.mods) {
    await window.api.saveLocalMods(data.mods)
  }

  message.success('导入成功')
}
```
