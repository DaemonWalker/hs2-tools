# 状态管理

本文档介绍 HS2-Tools 的 Zustand 状态管理方案。

## 概述

HS2-Tools 使用 [Zustand](https://github.com/pmndrs/zustand) 进行状态管理：

- **轻量级** - 仅 1KB 的代码
- **TypeScript 友好** - 完整的类型支持
- **无样板代码** - 简洁的 API
- **异步友好** - 天然支持异步操作
- **性能优化** - 支持 `useShallow` 避免不必要的重渲染

## Store 列表

| Store | 文件 | 用途 |
|-------|------|------|
| `settingStore` | `store/settingStore.ts` | 用户设置（路径、代理） |
| `modStore` | `store/modStore.ts` | Mod 数据和使用统计 |
| `downloadStore` | `store/downloadStore.ts` | 下载任务状态 |
| `themeStore` | `store/themeStore.ts` | 主题设置 |
| `sideloadStore` | `store/sideloadStore.ts` | Sideloader 数据和加载状态 |
| `taskStore` | `store/taskStore.ts` | 扫描和更新任务状态 |

## 设计原则

### 1. 单一职责

每个 Store 只管理一类状态：

```typescript
// ✅ 好的设计 - taskStore 只管理任务状态
interface TaskState {
  scanStatus: ScanStatus
  scanProgress: ScanProgress
  sideloadUpdateStatus: SideloadUpdateStatus
  sideloadUpdateProgress: SideloadUpdateProgress
  // 只包含任务相关状态
}

// ❌ 避免混合
interface BadState {
  scanStatus: ScanStatus      // 应该放在 taskStore
  settings: SettingModel      // 应该放在 settingStore
  mods: ModModel              // 应该放在 modStore
}
```

### 2. Selector Hooks 优化性能

使用 `useShallow` 和 selector hooks 避免不必要的重渲染：

```typescript
// ✅ 好的设计 - 使用 selector hooks
export const useModsSelector = () => useModStore(useShallow(state => state.mods))
export const useModUsageSelector = () => useModStore(useShallow(state => state.useage))

// 在组件中使用
const mods = useModsSelector()  // 只在 mods 变化时重渲染
```

### 3. 异步操作在 Store 中处理

```typescript
interface ModState {
  mods: ModModel
  useage: ModUseageModel

  // ✅ 在 Store 中处理异步
  setMods: (mods: ModModel) => Promise<void>
  setUseage: (useage: ModUseageModel) => Promise<void>
}

export const useModStore = create<ModState>((set) => ({
  mods: {},
  useage: {},

  setMods: async (mods) => {
    await saveLocalMods(mods)  // 持久化到存储
    set(() => ({ mods }))
  },

  setUseage: async (useage) => {
    await saveModUseage(useage)  // 持久化到存储
    set(() => ({ useage }))
  }
}))
```

## Store 详解

### 设置 Store (settingStore.ts)

```typescript
// src/renderer/src/store/settingStore.ts
import { create } from 'zustand'
import { useShallow } from 'zustand/shallow'

interface ModState {
  settings: SettingModel
  init: () => Promise<void>
  setPath: (path: string) => Promise<void>

  // 派生路径方法
  scenePath: () => string
  charaFemalePath: () => string
  modsPath: () => string
}

export const useSettingStore = create<SettingState>((set, get) => ({
  settings: { path: undefined },

  init: async () => {
    const settings = await getSettings()
    set({ settings: settings || { path: undefined } })
  },

  setPath: async (path) => {
    const settings = { ...get().settings, path }
    await saveSettings(settings)
    set({ settings })
  },

  // 派生路径
  scenePath: () => get().settings.path + '/UserData/Studio/scene',
  charaFemalePath: () => get().settings.path + '/UserData/chara/female',
  modsPath: () => get().settings.path + '/mods'
}))

// Selector hooks
export const useSettingsSelector = () => useSettingStore(useShallow(state => state.settings))
export const usePathSelector = () => useSettingStore(useShallow(state => state.settings.path))
```

```typescript
interface ModState {
```

### Sideload Store (sideloadStore.ts)

```typescript
// src/renderer/src/store/sideloadStore.ts
import { SideloadModel } from '@shared/models/sideloadModel'
import { create } from 'zustand'
import { useShallow } from 'zustand/shallow'
import ipcHandler from '@renderer/logic/ipcUtils'

interface SideloadState {
  sideload: SideloadModel      // { [guid: string]: url }
  isLoading: boolean           // 加载状态
  init: () => Promise<void>    // 初始化加载数据
  getDownloadUrl: (guid: string) => string | undefined
}

export const useSideloadStore = create<SideloadState>((set, get) => ({
  sideload: {},
  isLoading: false,
  init: async () => {
    set(() => ({ isLoading: true }))
    try {
      const sideload = await ipcHandler.initSideload()
      set(() => ({ sideload }))
    } finally {
      set(() => ({ isLoading: false }))
    }
  },
  getDownloadUrl: (guid: string) => {
    return get().sideload[guid]
  }
}))

// Selector hooks
export const useSideloadData = () => useSideloadStore(useShallow(state => state.sideload))
export const useGetDownloadUrl = () => useSideloadStore(state => state.getDownloadUrl)
```

**特点**:
- 管理从 betterrepack.com 获取的 sideload 数据
- 提供 `isLoading` 状态用于显示加载指示器
- 支持根据 GUID 快速查找下载 URL

### 任务 Store (taskStore.ts)

```typescript
// src/renderer/src/store/taskStore.ts
import { create } from 'zustand'
import { useShallow } from 'zustand/shallow'

export type ScanStatus = 'idle' | 'running' | 'complete' | 'error'
export type SideloadUpdateStatus = 'idle' | 'running' | 'success' | 'error'

interface ScanProgress {
  current: number
  info: {
    mod: string
    scene: string
    chara: string
  }
}

interface SideloadUpdateProgress {
  message?: string
  percent?: number
  current?: number
  error?: string
}

interface TaskState {
  // Scan 任务状态
  scanStatus: ScanStatus
  scanProgress: ScanProgress
  setScanStatus: (status: ScanStatus) => void
  setScanProgress: (progress: Partial<ScanProgress>) => void
  resetScan: () => void

  // Sideload 更新任务状态
  sideloadUpdateStatus: SideloadUpdateStatus
  sideloadUpdateProgress: SideloadUpdateProgress
  setSideloadUpdateStatus: (status: SideloadUpdateStatus) => void
  setSideloadUpdateProgress: (progress: SideloadUpdateProgress) => void
  resetSideloadUpdate: () => void
}

export const useTaskStore = create<TaskState>((set) => ({
  // Scan 初始状态
  scanStatus: 'idle',
  scanProgress: { current: -1, info: { mod: '', scene: '', chara: '' } },
  setScanStatus: (status) => set({ scanStatus: status }),
  setScanProgress: (progress) =>
    set((state) => ({ scanProgress: { ...state.scanProgress, ...progress } })),
  resetScan: () =>
    set({ scanStatus: 'idle', scanProgress: { current: -1, info: { mod: '', scene: '', chara: '' } } }),

  // Sideload 更新初始状态
  sideloadUpdateStatus: 'idle',
  sideloadUpdateProgress: {},
  setSideloadUpdateStatus: (status) => set({ sideloadUpdateStatus: status }),
  setSideloadUpdateProgress: (progress) =>
    set((state) => ({ sideloadUpdateProgress: { ...state.sideloadUpdateProgress, ...progress } })),
  resetSideloadUpdate: () =>
    set({ sideloadUpdateStatus: 'idle', sideloadUpdateProgress: {} })
}))

// Selector hooks
export const useScanStatus = () => useTaskStore(useShallow(state => ({
  status: state.scanStatus,
  progress: state.scanProgress,
  setStatus: state.setScanStatus,
  setProgress: state.setScanProgress,
  reset: state.resetScan
})))

export const useSideloadUpdateStatus = () => useTaskStore(useShallow(state => ({
  status: state.sideloadUpdateStatus,
  progress: state.sideloadUpdateProgress,
  setStatus: state.setSideloadUpdateStatus,
  setProgress: state.setSideloadUpdateProgress,
  reset: state.resetSideloadUpdate
})))
```

### 下载 Store (downloadStore.ts)

```typescript
// src/renderer/src/store/downloadStore.ts
import { create } from 'zustand'

export interface DownloadTask {
  guid: string
  name: string
  downloaded: number
  total: number
  speed: number
  percent: number
  status: 'pending' | 'downloading' | 'completed' | 'error'
  message?: string
}

interface DownloadState {
  totalCount: number
  currentIndex: number
  isBatchDownloading: boolean
  currentTask: DownloadTask | null
  tasks: Record<string, DownloadTask>

  startBatch: (total: number) => void
  startTask: (guid: string, name: string) => void
  updateProgress: (guid: string, progress: Partial<DownloadTask>) => void
  completeTask: (guid: string, success: boolean, message?: string) => void
  nextTask: () => void
  reset: () => void
}

export const useDownloadStore = create<DownloadState>((set) => ({
  totalCount: 0,
  currentIndex: 0,
  isBatchDownloading: false,
  currentTask: null,
  tasks: {},

  startBatch: (total) => set({
    totalCount: total,
    currentIndex: 0,
    isBatchDownloading: true,
    currentTask: null,
    tasks: {}
  }),

  startTask: (guid, name) => {
    const newTask: DownloadTask = {
      guid, name, downloaded: 0, total: -1,
      speed: 0, percent: 0, status: 'downloading'
    }
    set((state) => ({
      currentTask: newTask,
      tasks: { ...state.tasks, [guid]: newTask }
    }))
  },

  updateProgress: (guid, progress) => {
    set((state) => {
      const task = state.tasks[guid]
      if (!task) return state
      const updated = { ...task, ...progress }
      return {
        tasks: { ...state.tasks, [guid]: updated },
        currentTask: state.currentTask?.guid === guid ? updated : state.currentTask
      }
    })
  },

  completeTask: (guid, success, message) => {
    set((state) => {
      const task = state.tasks[guid]
      if (!task) return state
      const updated: DownloadTask = {
        ...task,
        status: success ? 'completed' : 'error',
        percent: success ? 100 : task.percent,
        message
      }
      return {
        tasks: { ...state.tasks, [guid]: updated },
        currentTask: state.currentTask?.guid === guid ? updated : state.currentTask
      }
    })
  },

  nextTask: () => set((state) => ({ currentIndex: state.currentIndex + 1 })),
  reset: () => set({ totalCount: 0, currentIndex: 0, isBatchDownloading: false, currentTask: null, tasks: {} })
}))
```

### 主题 Store (themeStore.ts)

```typescript
// src/renderer/src/store/themeStore.ts
import { create } from 'zustand'
import { persist } from 'zustand/middleware'

interface ThemeState {
  theme: 'modern-dark' | 'modern-light'
  setTheme: (theme: ThemeState['theme']) => void
  toggleTheme: () => void
}

export const useThemeStore = create(
  persist<ThemeState>(
    (set) => ({
      theme: 'modern-dark',
      setTheme: (theme) => {
        set({ theme })
        document.documentElement.setAttribute('data-theme', theme)
      },
      toggleTheme: () => set((state) => {
        const newTheme = state.theme === 'modern-dark' ? 'modern-light' : 'modern-dark'
        document.documentElement.setAttribute('data-theme', newTheme)
        return { theme: newTheme }
      })
    }),
    {
      name: 'theme-storage',
      getStorage: () => localStorage
    }
  )
)
```

## 在组件中使用

### 基础使用

```typescript
import { useSettingStore } from '@renderer/store/settingStore'

const MyComponent: FC = () => {
  // 只订阅需要的字段，避免不必要的重渲染
  const path = useSettingStore((state) => state.settings.path)
  const setPath = useSettingStore((state) => state.setPath)

  return (
    <div>
      <p>当前路径: {path}</p>
      <button onClick={() => setPath('new/path')}>
        修改路径
      </button>
    </div>
  )
}
```

```typescript
import { useModsSelector, useModUsageSelector } from '@renderer/store/modStore'

const ModStats: FC = () => {
```

### 在 Hooks 中使用

```typescript
// src/renderer/src/hooks/usePathSelector.ts
import { useState, useCallback } from 'react'
import { useSettingStore } from '@renderer/store/settingStore'

export const usePathSelector = (options?: { onSelect?: (path: string) => void }) => {
  const { setPath } = useSettingStore()
  const [loading, setLoading] = useState(false)

  const openSelectPath = useCallback(async () => {
    setLoading(true)
    try {
      const result = await window.api.selectPath()
      if (result && !result.canceled && result.path) {
        await setPath(result.path)
        options?.onSelect?.(result.path)
        return result.path
      }
    } finally {
      setLoading(false)
    }
    return null
  }, [setPath, options])

  return { openSelectPath, loading }
}
```

## 最佳实践

1. **选择性订阅** - 只订阅组件需要的字段，或使用 selector hooks
2. **使用 useShallow** - 对象类型状态使用 `useShallow` 避免深度比较问题
3. **异步错误处理** - 在 Store 中处理异步错误
4. **状态重置** - 提供 reset 方法清理状态
5. **TypeScript 类型** - 定义完整的 State 接口
6. **派生数据** - 使用 getter 方法或 selector 计算派生数据

## 注意事项

1. **Store 分离** - 按功能拆分 Store，避免一个 Store 过大
2. **状态更新** - 使用展开运算符创建新对象，避免直接修改原对象
3. **循环依赖** - 避免 Store 之间的循环依赖
4. **持久化** - 需要持久化的状态使用 `persist` 中间件
5. **性能优化** - 大数据列表使用 selector hooks 优化渲染性能
